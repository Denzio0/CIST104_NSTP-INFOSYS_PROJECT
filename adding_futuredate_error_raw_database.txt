DELIMITER $$

CREATE TRIGGER trg_attendance_date
BEFORE INSERT ON attendance
FOR EACH ROW
BEGIN
    IF NEW.session_date > CURDATE() THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Invalid future date';
    END IF;
END $$

DELIMITER ;
