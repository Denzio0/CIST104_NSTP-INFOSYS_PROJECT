SET FOREIGN_KEY_CHECKS = 0;
 
-- ── TABLE 1: students ────────────────────────────────────────────
CREATE TABLE IF NOT EXISTS students (
    student_id   INT            NOT NULL AUTO_INCREMENT,
    full_name    VARCHAR(150)   NOT NULL,
    program      ENUM('CWTS','ROTC') NOT NULL,
    year_level   VARCHAR(20)    NOT NULL,
    section      VARCHAR(20)    NOT NULL,
    contact_no   VARCHAR(20)    NULL,
    email        VARCHAR(100)   NULL,
    created_at   DATETIME       NOT NULL DEFAULT NOW(),
    PRIMARY KEY (student_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;
 
-- ── TABLE 2: users ──────────────────────────────────────────────
CREATE TABLE IF NOT EXISTS users (
    user_id       INT            NOT NULL AUTO_INCREMENT,
    username      VARCHAR(100)   NOT NULL,
    password_hash VARCHAR(255)   NOT NULL,
    role          ENUM('Admin','CWTS_Student','ROTC_Student','ROTC_Senior') NOT NULL,
    student_id    INT            NULL,
    is_active     TINYINT(1)     NOT NULL DEFAULT 1,
    last_login    DATETIME       NULL,
    PRIMARY KEY (user_id),
    UNIQUE KEY uq_username (username),
    CONSTRAINT fk_users_student
        FOREIGN KEY (student_id) REFERENCES students(student_id)
        ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;
 
-- ── TABLE 3: attendance ─────────────────────────────────────────
CREATE TABLE IF NOT EXISTS attendance (
    record_id    INT           NOT NULL AUTO_INCREMENT,
    student_id   INT           NOT NULL,
    session_name VARCHAR(100)  NOT NULL,
    session_date DATE          NOT NULL,
    status       ENUM('Present','Absent','Late','Excused') NOT NULL,
    remarks      TEXT          NULL,
    logged_by    INT           NULL,
    PRIMARY KEY (record_id),
    UNIQUE KEY uq_attendance_entry (student_id, session_name, session_date),
    KEY idx_attendance_session (session_name, session_date),
    CONSTRAINT fk_att_student FOREIGN KEY (student_id)
        REFERENCES students(student_id) ON DELETE CASCADE ON UPDATE CASCADE,
    CONSTRAINT fk_att_user FOREIGN KEY (logged_by)
        REFERENCES users(user_id) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;
 
-- ── TABLE 4: engagement ─────────────────────────────────────────
CREATE TABLE IF NOT EXISTS engagement (
    activity_id    INT             NOT NULL AUTO_INCREMENT,
    student_id     INT             NOT NULL,
    activity_name  VARCHAR(150)    NOT NULL,
    activity_date  DATE            NOT NULL,
    location       VARCHAR(150)    NULL,
    category       VARCHAR(80)     NULL,
    hours_rendered DECIMAL(5,2)    NOT NULL
                   CHECK (hours_rendered > 0 AND hours_rendered <= 24),
    verified       TINYINT(1)      NOT NULL DEFAULT 0,
    PRIMARY KEY (activity_id),
    CONSTRAINT fk_eng_student FOREIGN KEY (student_id)
        REFERENCES students(student_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;
 
-- ── TABLE 5: performance ────────────────────────────────────────
CREATE TABLE IF NOT EXISTS performance (
    entry_id     INT           NOT NULL AUTO_INCREMENT,
    student_id   INT           NOT NULL,
    drill_name   VARCHAR(100)  NOT NULL,
    drill_date   DATE          NOT NULL,
    rating       ENUM('Excellent','Proficient','Satisfactory',
                      'Needs_Improvement','Unsatisfactory') NOT NULL,
    remarks      TEXT          NULL,
    officer_name VARCHAR(100)  NULL,
    PRIMARY KEY (entry_id),
    CONSTRAINT fk_perf_student FOREIGN KEY (student_id)
        REFERENCES students(student_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;
 
-- ── TABLE 6: points ─────────────────────────────────────────────
CREATE TABLE IF NOT EXISTS points (
    transaction_id   INT     NOT NULL AUTO_INCREMENT,
    student_id       INT     NOT NULL,
    amount           INT     NOT NULL CHECK (amount > 0),
    type             ENUM('award','deduct') NOT NULL,
    reason           TEXT    NOT NULL,
    transaction_date DATE    NOT NULL,
    admin_id         INT     NULL,
    PRIMARY KEY (transaction_id),
    KEY idx_points_student (student_id),
    CONSTRAINT fk_pts_student FOREIGN KEY (student_id)
        REFERENCES students(student_id) ON DELETE CASCADE ON UPDATE CASCADE,
    CONSTRAINT fk_pts_admin FOREIGN KEY (admin_id)
        REFERENCES users(user_id) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;
 
-- ── TABLE 7: demerits ───────────────────────────────────────────
CREATE TABLE IF NOT EXISTS demerits (
    demerit_id   INT           NOT NULL AUTO_INCREMENT,
    student_id   INT           NOT NULL,
    reason       VARCHAR(200)  NOT NULL,
    severity     ENUM('Minor','Moderate','Major') NOT NULL,
    notes        TEXT          NULL,
    officer_name VARCHAR(100)  NULL,
    demerit_date DATE          NOT NULL,
    status       ENUM('Active','Appealed','Dismissed') NOT NULL DEFAULT 'Active',
    PRIMARY KEY (demerit_id),
    CONSTRAINT fk_dem_student FOREIGN KEY (student_id)
        REFERENCES students(student_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;
 
-- ── TABLE 8: announcements ──────────────────────────────────────
CREATE TABLE IF NOT EXISTS announcements (
    announcement_id INT           NOT NULL AUTO_INCREMENT,
    title           VARCHAR(200)  NOT NULL,
    body            TEXT          NOT NULL,
    audience        ENUM('All','CWTS','ROTC') NOT NULL DEFAULT 'All',
    priority        ENUM('Normal','Important','Urgent') NOT NULL DEFAULT 'Normal',
    posted_date     DATETIME      NOT NULL DEFAULT NOW(),
    posted_by       INT           NULL,
    is_pinned       TINYINT(1)    NOT NULL DEFAULT 0,
    PRIMARY KEY (announcement_id),
    KEY idx_ann_pinned (is_pinned, posted_date),
    CONSTRAINT fk_ann_user FOREIGN KEY (posted_by)
        REFERENCES users(user_id) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;
 
-- ── TABLE 9: notifications ──────────────────────────────────────
CREATE TABLE IF NOT EXISTS notifications (
    notif_id     INT           NOT NULL AUTO_INCREMENT,
    student_id   INT           NOT NULL,
    type         VARCHAR(80)   NOT NULL,
    message      TEXT          NOT NULL,
    created_date DATETIME      NOT NULL DEFAULT NOW(),
    is_read      TINYINT(1)    NOT NULL DEFAULT 0,
    PRIMARY KEY (notif_id),
    KEY idx_notif_read (is_read),
    CONSTRAINT fk_notif_student FOREIGN KEY (student_id)
        REFERENCES students(student_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;
 
-- ── TABLE 10: disputes ──────────────────────────────────────────
CREATE TABLE IF NOT EXISTS disputes (
    dispute_id       INT     NOT NULL AUTO_INCREMENT,
    student_id       INT     NOT NULL,
    record_id        INT     NOT NULL,
    reason           TEXT    NOT NULL,
    requested_status ENUM('Present','Excused','Late') NOT NULL,
    submitted_date   DATE    NOT NULL,
    status           ENUM('Pending','Approved','Rejected') NOT NULL DEFAULT 'Pending',
    admin_response   TEXT    NULL,
    PRIMARY KEY (dispute_id),
    CONSTRAINT fk_disp_student FOREIGN KEY (student_id)
        REFERENCES students(student_id) ON DELETE CASCADE ON UPDATE CASCADE,
    CONSTRAINT fk_disp_record FOREIGN KEY (record_id)
        REFERENCES attendance(record_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;
 
-- ── TABLE 11: read_announcements ────────────────────────────────
CREATE TABLE IF NOT EXISTS read_announcements (
    student_id      INT      NOT NULL,
    announcement_id INT      NOT NULL,
    read_date       DATETIME NOT NULL DEFAULT NOW(),
    PRIMARY KEY (student_id, announcement_id),
    CONSTRAINT fk_ra_student FOREIGN KEY (student_id)
        REFERENCES students(student_id) ON DELETE CASCADE ON UPDATE CASCADE,
    CONSTRAINT fk_ra_announcement FOREIGN KEY (announcement_id)
        REFERENCES announcements(announcement_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;
 
-- ── TABLE 12: calendar_events ───────────────────────────────────
CREATE TABLE IF NOT EXISTS calendar_events (
    event_id      INT           NOT NULL AUTO_INCREMENT,
    title         VARCHAR(200)  NOT NULL,
    event_date    DATE          NOT NULL,
    program_type  ENUM('All','CWTS','ROTC') NOT NULL DEFAULT 'All',
    location      VARCHAR(150)  NULL,
    description   TEXT          NULL,
    reminder_flag TINYINT(1)    NOT NULL DEFAULT 0,
    created_by    INT           NULL,
    PRIMARY KEY (event_id),
    KEY idx_calendar_date (event_date),
    CONSTRAINT fk_cal_user FOREIGN KEY (created_by)
        REFERENCES users(user_id) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;
 
-- ── TABLE 13: feedback ──────────────────────────────────────────
CREATE TABLE IF NOT EXISTS feedback (
    feedback_id    INT           NOT NULL AUTO_INCREMENT,
    student_id     INT           NOT NULL,
    submitted_date DATETIME      NOT NULL DEFAULT NOW(),
    type           VARCHAR(80)   NOT NULL,
    subject        VARCHAR(200)  NOT NULL,
    body           TEXT          NOT NULL,
    status         ENUM('Open','Resolved') NOT NULL DEFAULT 'Open',
    admin_response TEXT          NULL,
    PRIMARY KEY (feedback_id),
    CONSTRAINT fk_fb_student FOREIGN KEY (student_id)
        REFERENCES students(student_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;
 
SET FOREIGN_KEY_CHECKS = 1;
